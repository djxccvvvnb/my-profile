size_t capacity; size_t item_size; } Vec; Vec vec_new(size_t item_size) { Vec v; v.item_size = item_size; v.capacity = 4; v.count = 0; v.items = malloc(item_size * v.capacity); return v; } void vec_push(Vec *v, const void *item) { if (v->count >= v->capacity) { v->capacity *= 2; v->items = realloc(v->items, v->item_size * v->capacity); } memcpy((char*)v->items + v->count * v->item_size, item, v->item_size); v->count++; } void vec_free(Vec *v) { free(v->items); } /* --- 2. 字符串哈希表 --- */ #define HASH_CAP 256 typedef struct Entry { char *key; int value; struct Entry *next; } Entry; typedef struct { Entry *buckets[HASH_CAP]; int size; } HashTable; static unsigned hash(const char *s) { unsigned h = 1469598103U; while (*s) { h ^= *(unsigned char*)s++; h *= 1099511628211U; } return h; } HashTable ht_new(void) { HashTable t; memset(t.buckets, 0, sizeof(t.buckets)); t.size = 0; return t; } void ht_put(HashTable *t, const char *key, int val) { unsigned idx = hash(key) % HASH_CAP; Entry *e = t->buckets[idx]; while (e) { if (strcmp(e->key, key)==0) { e->value = val; return; } e = e->next; } e = malloc(sizeof(Entry)); e->key = strdup(key); e->value = val; e->next = t->buckets[idx]; t->buckets[idx] = e; t->size++; } int ht_get(HashTable *t, const char *key, int *out) { unsigned idx = hash(key) % HASH_CAP; for (Entry *e = t->buckets[idx]; e; e = e->next) if (strcmp(e->key, key)==0) { *out = e->value; return 1; } return 0; } void ht_free(HashTable *t) { for (int i=0; i<HASH_CAP; i++) for (Entry *e=t->buckets[i], *n; e; e=n) { n=e->next; free(e->key); free(e); } } /* --- 3. 改进的冒泡排序 --- */ void optimized_bubble(int *arr, int n) { for (int pass=0; pass<n-1; pass++) { int swapped = 0; for (int j=0; j<n-1-pass; j++) if (arr[j] > arr[j+1]) { int tmp=arr[j]; arr[j]=arr[j+1]; arr[j+1]=tmp; swapped=1; } if (!swapped) break; } } /* --- 4. 位运算：汉明权重 --- */ int hamming_weight(uint32_t n) { int cnt=0;
