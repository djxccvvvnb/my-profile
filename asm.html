Assembly (ASM) — x86-64 底层原理

Registers · Stack · Syscalls · Bit Operations

; 董健贤 &middot; x86-64 Assembly 实战合集 ; —— 系统调用、栈操作、位运算、字符串处理 ; 使用 nasm 语法，Linux x86-64 环境编译: ; nasm -f elf64 main.asm && ld main.o -o main section .data msg_hello db 'Hello, World!', 0x0a msg_hello_len equ $ - msg_hello section .bss buf resb 32 section .text global _start ; ========================================= ; 1. 打印字符串 (sys_write) ; ========================================= print_string: push rbp mov rbp, rsp mov rax, 1 ; sys_write mov rdi, 1 ; stdout mov rsi, [rbp+8] ; buffer mov rdx, [rbp+16] ; length syscall pop rbp ret ; ========================================= ; 2. 退出程序 (sys_exit) ; ========================================= exit_program: push rbp mov rbp, rsp mov rax, 60 ; sys_exit mov rdi, [rbp+8] ; exit code syscall pop rbp ret ; ========================================= ; 3. 位运算：计算汉明权重 (popcnt) ; ========================================= popcnt32: mov eax, edi xor ecx, ecx ; 计数器清零 bitloop: test eax, eax jz done_popcnt and eax, eax - 1 ; Brian Kernighan 算法 inc ecx jmp bitloop done_popcnt: mov rax, rcx ret ; ========================================= ; 4. 简单冒泡排序 ; ========================================= bubble_sort: push rbp mov rbp, rsp push rbx push rcx push rdx push r8 push r9 mov rcx, rsi ; n dec_n: dec rcx jle sort_done mov r8, rcx xor r9, r9 ; swapped = 0 inner: cmp r9, r8 jge outer lea rdx, [rdi + r9*4] mov eax, [rdx] mov ebx, [rdx + 4] cmp eax, ebx jle no_swap xchg eax, ebx mov [rdx], eax mov [rdx + 4], ebx mov byte [r9], 1 ; swapped = 1 no_swap: inc r9 jmp inner outer: test byte [r9], 1 jz sort_done jmp dec_n sort_done: pop r9 pop r8 pop rdx pop rcx pop rbx pop rbp ret ; ========================================= ; 主函数入口 ; ========================================= _start: lea rdi, [rel msg_hello] mov rsi, msg_hello_len call print_string mov edi, 171 call popcnt32 lea rdi, [rel numbers] mov rsi, 7 call bubble_sort mov rdi, 0 call exit_program section .data numbers dd 64, 34, 25, 12, 22, 11, 90
