# 董健贤 &middot; Python 高级实战 from typing import Callable, TypeVar, Any from functools import wraps import time T = TypeVar("T") # --- 性能计时装饰器 --- def timer(func: Callable[..., T]) -> Callable[..., T]: @wraps(func) def wrapper(*args, **kwargs) -> T: t0 = time.perf_counter() result = func(*args, **kwargs) elapsed = time.perf_counter() - t0 print(f"[{func.__name__}] executed in {elapsed:.4f}s") return result return wrapper # --- 记忆化装饰器 --- def memoize(func: Callable[..., T]) -> Callable[..., T]: cache: dict[tuple, T] = {} @wraps(func) def wrapper(*args): key = args if key not in cache: cache[key] = func(*args) return cache[key] return wrapper # --- LRU Cache --- class DLNode: __slots__ = ("key", "val", "prev", "next") def __init__(self, key, val): self.key = key self.val = val self.prev = None self.next = None class DoublyLinkedList: def __init__(self): self.head = DLNode(None, None) self.tail = DLNode(None, None) self.head.next = self.tail self.tail.prev = self.head self._map: dict[int, DLNode] = {} def append_right(self, node: DLNode) -> None: pred, succ = self.tail.prev, self.tail pred.next = succ.prev = node node.prev = pred node.next = succ def remove(self, node: DLNode) -> None: pred, succ = node.prev, node.next pred.next = succ succ.prev = pred def pop_left(self) -> DLNode | None: if self.head.next == self.tail: return None node = self.head.next self.remove(node) return node class LRUCache: def __init__(self, capacity: int): self.cap = capacity self.dll = DoublyLinkedList() self.map: dict[int, DLNode] = {} def get(self, key: int) -> int: if key not in self.map: return -1 node = self.map[key] self.dll.remove(node) self.dll.append_right(node) return node.val def put(self, key: int, val: int) -> None: if key in self.map: node = self.map[key] node.val = val self.dll.remove(node) self.dll.append_right(node) else: if len(self.map) >= self.cap: oldest = self.dll.pop_left() del self.map[oldest.key] node = DLNode(key, val) self.dll.append_right(node) self.map[key] = node if __name__ == "__main__": cache = LRUCache(3) cache.put(1, 10) cache.put(2, 20) cache.put(3, 30) print(cache.get(1)) # 10 print(cache.get(4)) # -1
